# hrms-workflow（入转调离 & 审批中心）建表脚本

> **来源**：[入转调离 & 审批中心-后端系分.md](./入转调离%20&%20审批中心-后端系分.md) §7  
> **模块**：`hrms-workflow`  
> **本模块表数量**：11 张（表名无前缀）  
> **说明**：脚本与系分 §7.2～§7.12 DDL 一致；索引含 CREATE 内 KEY，并参照 §7.13。

```sql
-- ============================================================
-- hrms-workflow · 入转调离 & 审批中心（11 张表）
-- ============================================================

-- 1. 流程定义
CREATE TABLE approval_process_def (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    process_type    VARCHAR(32) NOT NULL COMMENT 'ONBOARDING/REGULARIZATION/TRANSFER/RESIGNATION',
    name            VARCHAR(64) NOT NULL,
    nodes_json      JSON        NOT NULL COMMENT '审批节点配置',
    sla_hours       INT         NOT NULL DEFAULT 48,
    status          TINYINT     NOT NULL DEFAULT 1,
    UNIQUE KEY uk_type (process_type)
) COMMENT='流程定义';

-- 2. 审批实例
CREATE TABLE approval_instance (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    process_type    VARCHAR(32) NOT NULL,
    business_key    VARCHAR(64) NOT NULL COMMENT '业务表主键',
    status          VARCHAR(32) NOT NULL,
    initiator_id    BIGINT      NOT NULL,
    current_node    INT         NOT NULL DEFAULT 1,
    created_at      DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_type_status (process_type, status),
    KEY idx_initiator (initiator_id),
    KEY idx_business (process_type, business_key)
) COMMENT='审批实例';

-- 3. 审批任务（待办）；API 响应将 sla_deadline 映射为 dueAt
CREATE TABLE approval_task (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id     BIGINT      NOT NULL,
    node_order      INT         NOT NULL,
    assignee_id     BIGINT      NOT NULL COMMENT '原审批人',
    actual_assignee_id BIGINT   NULL COMMENT '委托解析后的实际审批人',
    status          VARCHAR(16) NOT NULL COMMENT 'PENDING/APPROVED/REJECTED/FORWARDED',
    comment         VARCHAR(512) NULL,
    sla_deadline    DATETIME     NULL,
    overdue         TINYINT      NOT NULL DEFAULT 0,
    completed_at    DATETIME     NULL,
    KEY idx_assignee_status (assignee_id, status),
    KEY idx_actual_assignee (actual_assignee_id, status),
    KEY idx_instance (instance_id)
) COMMENT='审批任务';

-- 4. 审批流转日志
CREATE TABLE approval_log (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id     BIGINT      NOT NULL,
    task_id         BIGINT      NULL,
    operator_id     BIGINT      NOT NULL COMMENT '实际操作人',
    on_behalf_of_id BIGINT      NULL COMMENT '被代审人',
    display_text    VARCHAR(128) NULL COMMENT '孙强 代 李明 审批',
    action          VARCHAR(16) NOT NULL COMMENT 'SUBMIT/APPROVE/REJECT/WITHDRAW/FORWARD',
    comment         VARCHAR(512) NULL,
    from_status     VARCHAR(32) NULL,
    to_status       VARCHAR(32) NULL,
    created_at      DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_instance (instance_id)
) COMMENT='审批流转日志';

-- 5. 委托审批（同一 delegator 仅一条 ACTIVE）
CREATE TABLE approval_delegation (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    delegator_id    BIGINT      NOT NULL,
    delegate_user_id BIGINT     NOT NULL,
    start_date      DATE        NOT NULL,
    end_date        DATE        NOT NULL,
    reason          VARCHAR(256) NULL,
    status          VARCHAR(16) NOT NULL DEFAULT 'ACTIVE' COMMENT 'ACTIVE/CANCELLED',
    cancelled_at    DATETIME     NULL,
    created_at      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_delegator (delegator_id, status),
    KEY idx_delegate (delegate_user_id, status, start_date, end_date)
) COMMENT='委托审批，同一 delegator 仅一条 ACTIVE';

-- 6. 入职申请
CREATE TABLE onboarding_application (
    id                      BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id             BIGINT       NULL,
    status                  VARCHAR(16)  NOT NULL COMMENT 'DRAFT/APPROVING/APPROVED/REJECTED/ONBOARDED/ABANDONED',
    name                    VARCHAR(32)  NOT NULL,
    gender                  VARCHAR(8)   NOT NULL COMMENT 'MALE/FEMALE，API 映射 male/female',
    mobile                  VARCHAR(16)  NOT NULL,
    email                   VARCHAR(128) NOT NULL,
    id_number_enc           VARCHAR(256) NOT NULL COMMENT '身份证密文',
    id_number_hash          VARCHAR(64)  NOT NULL COMMENT 'SHA256检索',
    expected_onboard_date   DATE         NOT NULL,
    department_id           BIGINT       NOT NULL,
    position_id             BIGINT       NOT NULL,
    employment_type         VARCHAR(16)  NOT NULL COMMENT 'fulltime/parttime/intern，API 字段 employmentType',
    probation_months        INT          NOT NULL,
    probation_salary_ratio  DECIMAL(5,4) NOT NULL,
    manager_id              BIGINT       NULL,
    employee_id             BIGINT       NULL COMMENT '审批通过后关联',
    created_by              BIGINT       NOT NULL,
    created_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_status (status),
    KEY idx_dept (department_id),
    KEY idx_mobile (mobile)
) COMMENT='入职申请';

-- 7. 转正申请
CREATE TABLE regularization_application (
    id                      BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id             BIGINT       NULL,
    employee_id             BIGINT       NOT NULL,
    status                  VARCHAR(16)  NOT NULL,
    probation_start_date    DATE         NOT NULL,
    probation_end_date      DATE         NOT NULL,
    performance_evaluation  TEXT         NOT NULL,
    salary_adjustment       DECIMAL(12,2) NULL,
    approval_result         VARCHAR(16)  NULL COMMENT 'PASS/EXTEND/FAIL',
    extend_months           INT          NULL,
    created_by              BIGINT       NOT NULL,
    created_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_employee (employee_id)
) COMMENT='转正申请';

-- 8. 调岗申请
CREATE TABLE transfer_application (
    id                      BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id             BIGINT       NULL,
    employee_id             BIGINT       NOT NULL,
    status                  VARCHAR(16)  NOT NULL,
    from_department_id      BIGINT       NOT NULL,
    new_department_id       BIGINT       NOT NULL,
    new_position_id         BIGINT       NULL,
    new_job_level           VARCHAR(8)   NULL,
    new_manager_id          BIGINT       NULL,
    salary_adjustment       DECIMAL(12,2) NULL,
    effective_date          DATE         NOT NULL,
    reason                  VARCHAR(512) NULL,
    created_by              BIGINT       NOT NULL,
    created_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_employee (employee_id)
) COMMENT='调岗申请';

-- 9. 调岗历史
CREATE TABLE employee_transfer_history (
    id                  BIGINT PRIMARY KEY AUTO_INCREMENT,
    employee_id         BIGINT       NOT NULL,
    transfer_app_id     BIGINT       NOT NULL,
    from_department_id  BIGINT       NOT NULL,
    to_department_id    BIGINT       NOT NULL,
    from_position_id    BIGINT       NULL,
    to_position_id      BIGINT       NULL,
    transfer_date       DATE         NOT NULL,
    reason              VARCHAR(512) NULL,
    created_at          DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_employee (employee_id)
) COMMENT='调岗历史';

-- 10. 员工门户离职申请
CREATE TABLE employee_resignation_request (
    id                  BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id         BIGINT       NULL,
    employee_id         BIGINT       NOT NULL,
    status              VARCHAR(16)  NOT NULL COMMENT 'PENDING/APPROVED/REJECTED/CANCELLED',
    expected_resign_date DATE        NOT NULL,
    reason_category     VARCHAR(16)  NOT NULL COMMENT 'VOLUNTARY/INVOLUNTARY/NEGOTIATED',
    resignation_type    VARCHAR(16)  NOT NULL COMMENT 'RESIGN/DISMISS/CONTRACT_EXPIRE/OTHER',
    reason_detail       VARCHAR(512) NULL,
    created_at          DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_employee (employee_id),
    KEY idx_status (status)
) COMMENT='员工离职申请';

-- 11. HR 正式离职
CREATE TABLE resignation_application (
    id                      BIGINT PRIMARY KEY AUTO_INCREMENT,
    instance_id             BIGINT       NULL,
    request_id              BIGINT       NOT NULL COMMENT '关联 employee_resignation_request.id',
    employee_id             BIGINT       NOT NULL,
    status                  VARCHAR(16)  NOT NULL COMMENT 'APPROVING/PENDING_RESIGN/REJECTED/RESIGNED',
    resignation_date        DATE         NOT NULL,
    reason_category         VARCHAR(16)  NOT NULL COMMENT 'VOLUNTARY/INVOLUNTARY/NEGOTIATED',
    resignation_type        VARCHAR(16)  NOT NULL COMMENT 'RESIGN/DISMISS/CONTRACT_EXPIRE/OTHER',
    reason_detail           VARCHAR(512) NULL,
    handover_employee_id    BIGINT       NOT NULL,
    created_by              BIGINT       NOT NULL,
    created_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    KEY idx_employee (employee_id),
    KEY idx_request (request_id),
    KEY idx_resign_date (resignation_date, status)
) COMMENT='HR正式离职申请';
```

## 依赖的其他模块表

> 以下表**不在本模块建表脚本内**，由 A/B/D 等模块维护；本模块通过外键语义字段、审批人解析或 Feign/MQ 读写。依据系分 §2.2、§7.2 审批人解析、§9.5 跨模块联动整理。

### A 模块 · 权限 + 组织架构（hrms-auth / hrms-org）

| 表名 | 依赖原因 |
| --- | --- |
| `department` | 入职/调岗单据写 `department_id` / `from_department_id` / `new_department_id`；审批人 `DEPT_MANAGER` / `NEW_DEPT_MANAGER` 解析读 `head_employee_id`；无负责人则 submit 拒绝（Q-C-07） |
| `position` | 入职写 `position_id`；调岗可选 `new_position_id`；二审条件依赖是否标准职位 / 职级薪资范围 |
| `sys_user` | 入职审批通过后创建登录账号；离职生效后禁用账号；审批人/发起人 ID 对应账号侧用户 |
| `sys_role`（及用户-角色关联表） | 审批节点 `assigneeType=ROLE`（如 `HR_STAFF`）按角色解析在职审批人 |

### B 模块 · 员工档案 + 个人中心（hrms-employee）

| 表名 | 归属能力 | 依赖原因 |
| --- | --- | --- |
| `employee` | 员工档案 | 转正/调岗/离职单据关联 `employee_id`；入职审批通过后创建员工主档；转正更新 `employment_status`；调岗更新部门/岗位/上级；离职置 `PENDING_RESIGN` / `RESIGNED` |
| `employee_personal` | 员工档案 | 入职审批通过建档时写入身份证、地址等个人信息 |
| `employee_contract` | 员工档案 | 入职联动写入合同与约定薪资等；与申请单用工类型、试用比例对应 |
| `employee_bank` | 员工档案 | 建档时可选落银行卡信息（档案侧维护，本模块不建表） |
| `employee_no_history`（或工号序号相关表） | 员工档案 | 离职生效「释放工号」由员工档案侧处理，本模块触发联动 |
| 个人中心读路径依赖的 `employee` / `employee_personal` | 个人中心 | 门户「我的档案」只读展示；本模块不写个人中心表，但离职申请发起人、交接人等均指向员工主档 |
| `employee_mobile_change_application` | 个人中心 | 手机号变更走档案侧审批（`MOBILE_CHANGE`），**非本模块建表**；审批引擎若复用公共能力时与本模块流程表隔离 |

> 说明：调岗历史表 `employee_transfer_history`、员工离职申请表 `employee_resignation_request` **由本模块建表并写入**；调岗/离职生效时同步更新的是 B 模块的 `employee`（及档案扩展表）状态与组织信息。

### D 模块 · 考勤请假 + 薪资

#### 考勤请假

| 表名 | 依赖原因 |
| --- | --- |
| `attendance_group` | 考勤组主数据；入职到岗 / 离职生效时按员工归属部门匹配考勤组 |
| `attendance_group_member` | 入职 `onboarded` 后经 MQ `hrms.employee.event` **写入成员（关联考勤组）**；离职生效 **删除/移出成员** |
| `attendance_group_scope` | 考勤组适用部门/范围配置，解析员工应入组时只读依赖 |
| `workday_config` | 工作日/班次配置（考勤组侧）；入组后员工打卡与自然日结算依赖，本模块不直接写 |
| `attendance_record` | 打卡明细；员工入职生效后产生，离职日之后不再纳入有效出勤（由考勤模块按状态处理） |
| `attendance_daily_summary` / `attendance_monthly_summary` | 日/月汇总；离职结算、算薪取数会读到对应员工汇总 |
| `attendance_month_lock` | 考勤月锁定；离职当月是否已锁定影响考勤与算薪边界（D 模块规则） |
| `leave_balance` | 请假余额；离职生效后余额冻结/清算由请假模块处理 |
| `leave_application` | 请假单；在职期间有效，离职后不再发起（状态以 `employee` 为准） |
| `attendance_supplement` | 补卡申请；同上，依赖在职员工 |

#### 薪资

| 表名 | 依赖原因 |
| --- | --- |
| `employee_salary_profile` | 入职到岗「占位薪资档案」：尚无完整薪资方案时创建档案占位，供后续算薪 |
| `employee_salary_history` | 调岗/转正若有调薪，写入薪资变更历史（与档案联动） |
| `payroll_scheme` / `payroll_scheme_scope` / `payroll_scheme_item` | 薪资方案及适用范围；占位档案、入职选账套时只读依赖 |
| `payroll_batch` | 算薪批次；离职生效后「薪资结算至离职日」纳入批次范围 |
| `payroll_detail` | 员工算薪明细；离职当月按离职日截断计薪 |
| `payroll_adjustment` | 离职结算可能产生补扣/调整项（由薪资模块维护） |

### 逻辑外键一览（本模块字段 → 他模块 / 本模块表）

| 本模块字段 | 指向 |
| --- | --- |
| `*_application.instance_id` | 本模块 `approval_instance.id` |
| `approval_task.instance_id` / `approval_log.instance_id` | 本模块 `approval_instance.id` |
| `onboarding_application.department_id` / `position_id` | `department` / `position` |
| `onboarding_application.employee_id`、`manager_id` | `employee`（审批通过后 / 直属上级） |
| `*_application.employee_id`、`handover_employee_id`、`new_manager_id` | `employee` |
| `from_department_id` / `new_department_id` / `to_department_id` | `department` |
| `new_position_id` / `from_position_id` / `to_position_id` | `position` |
| `resignation_application.request_id` | 本模块 `employee_resignation_request.id` |
| `assignee_id` / `initiator_id` / `delegator_id` / `operator_id` 等 | 通常对应 `sys_user` / 员工账号体系 |
